    function loadGroup(name, defaultImages) {
        try {
            const stored = localStorage.getItem(STORAGE_PREFIX + name);
            if (stored) {
                const arr = JSON.parse(stored);
                if (Array.isArray(arr)) return arr.filter(item => item && item.url);
            }
        } catch (e) {}
        return [...defaultImages];
    }

        try { localStorage.setItem(STORAGE_PREFIX + name, JSON.stringify(groups[name])); } catch (e) {}

    // ==================== 编辑器插入逻辑 ====================
    function triggerInput(el) {
        el.dispatchEvent(new Event('input', { bubbles: true }));
        el.dispatchEvent(new Event('change', { bubbles: true }));

    function getCodeMirror() {
        const cmEl = document.querySelector('.CodeMirror');
        if (cmEl?.CodeMirror) return cmEl.CodeMirror;
        if (window.easyMDE) return window.easyMDE.codemirror;
        return null;
    }

    function getTextarea() { return document.querySelector('textarea[name="body"]'); }

    function findHiddenEditor() {
        const selector = 'div[style*="visibility: hidden"][style*="position: absolute"][style*="overflow: scroll"]';
        for (const el of document.querySelectorAll(selector)) {
            if (el.offsetWidth > 0 || el.offsetHeight > 0 || el.textContent.length > 0) return el;
        }
        return null;
    }

    function insertMarkdown(text) {
        // 1. CodeMirror
        const cm = getCodeMirror();
        if (cm) {
            cm.focus();
            cm.replaceSelection(text);
            triggerInput(cm.getInputField());
            const ta = getTextarea();
            if (ta) triggerInput(ta);
            return true;
        }

        // 2. 隐藏 div
        const hiddenDiv = findHiddenEditor();
        if (hiddenDiv) {
            hiddenDiv.focus();
            const selection = window.getSelection();
            if (selection?.rangeCount > 0) {
                const range = selection.getRangeAt(0);
                if (hiddenDiv.contains(range.commonAncestorContainer)) {
                    range.deleteContents();
                    const textNode = document.createTextNode(text);
                    range.insertNode(textNode);
                    range.setStartAfter(textNode);
                    range.setEndAfter(textNode);
                    selection.removeAllRanges();
                    selection.addRange(range);
                    triggerInput(hiddenDiv);
                    return true;
                }
            }
            hiddenDiv.appendChild(document.createTextNode(text));
            triggerInput(hiddenDiv);
            return true;
        }

        // 3. textarea
        const textarea = getTextarea();
        if (textarea) {
            const start = textarea.selectionStart ?? textarea.value.length;
            const end = textarea.selectionEnd ?? start;
            textarea.value = textarea.value.slice(0, start) + text + textarea.value.slice(end);
            textarea.selectionStart = textarea.selectionEnd = start + text.length;
            textarea.focus();
            triggerInput(textarea);
            return true;
        }

        console.warn('[表情包工具] 未找到任何可插入的编辑器区域');
        return false;
    }

        });
    }

    // 渲染图片网格
    function renderGrid() {
        gridContainer.innerHTML = '';
        currentImages.forEach((item, index) => {
            const li = document.createElement('li');
            li.style.cssText = 'position:relative;width:56px;height:56px;border-radius:8px;overflow:hidden;cursor:pointer;border:1px solid #ffffff36';

            const img = document.createElement('img');
            img.src = item.url;
            img.title = item.desc || '图片描述';
            img.alt = item.desc || '图片描述';
            img.style.cssText = 'width:100%;height:100%;object-fit:cover;display:block';
            img.addEventListener('click', (e) => {
                e.stopPropagation();
                insertMarkdown(`![${item.desc || '图片描述'}](${item.url})`);
                hidePanel();
            });

            const delBtn = document.createElement('span');
            delBtn.textContent = '✕';
            delBtn.title = '删除';
            delBtn.style.cssText = 'position:absolute;top:2px;right:2px;width:18px;height:18px;background:rgba(0,0,0,0.6);color:#fff;border-radius:50%;display:none;align-items:center;justify-content:center;font-size:12px;line-height:1;cursor:pointer;z-index:2';
            delBtn.addEventListener('click', (e) => {
                e.stopPropagation();
                currentImages.splice(index, 1);
                saveGroup(currentGroupName);
                renderGrid();
                showToast('已删除');
            });

            li.append(img, delBtn);
            li.addEventListener('mouseenter', () => delBtn.style.display = 'flex');
            li.addEventListener('mouseleave', () => delBtn.style.display = 'none');
            gridContainer.appendChild(li);
        });
    }

    function hidePanel() {
        panel.style.display = 'none';
        if (addArea) addArea.style.display = 'none';
    }

    function togglePanel(btn) {
        const p = createPanel();
        renderTabs();
        renderGrid();
        if (p.style.display === 'block') return hidePanel();

        p.style.display = 'block';
        const btnRect = btn.getBoundingClientRect();
        const panelRect = p.getBoundingClientRect();

        // 右对齐
        let left = btnRect.right + scrollX - panelRect.width;
        let top = btnRect.top + scrollY - panelRect.height - 8;

        if (left < scrollX + 8) left = scrollX + 8;
        if (left + panelRect.width > scrollX + innerWidth - 8) left = scrollX + innerWidth - panelRect.width - 8;
        if (top < scrollY + 8) top = btnRect.bottom + scrollY + 8;

        p.style.left = left + 'px';
        p.style.top = top + 'px';
    }

    // ==================== 编辑器按钮 ====================
    function createEditorButton(emojiBtn) {
        if (newBtn) return newBtn;

        newBtn = emojiBtn.cloneNode(false);
        newBtn.removeAttribute('data-nb-editor-emoji');
        newBtn.title = '表情包';
        newBtn.setAttribute('aria-label', '表情包');
        newBtn.innerHTML = EDITOR_SVG;
        newBtn.addEventListener('click', (e) => {
            e.preventDefault();
            e.stopPropagation();
            togglePanel(newBtn);
        });
        return newBtn;
    }

    // ==================== 帖子图片收集（添加到“我的”组） ====================
    function initPostImageCollector() {
        document.addEventListener('mouseover', (e) => {
            const img = e.target.closest?.('.post-item.post-entry .post-content img');
            if (!img || img.dataset.emojiCollector) return;

            img.dataset.emojiCollector = '1';
            const parent = img.parentElement;
            if (!parent) return;

            if (getComputedStyle(parent).position === 'static') parent.style.position = 'relative';

            const addIcon = document.createElement('span');
            addIcon.textContent = '+';
            addIcon.title = '添加到我的表情包';
            addIcon.style.cssText = `position:absolute;top:${img.offsetTop + 4}px;left:${img.offsetLeft + 4}px;width:24px;height:24px;background:rgba(74,144,217,0.9);color:#fff;border-radius:50%;display:flex;align-items:center;justify-content:center;font-size:18px;cursor:pointer;z-index:10;box-shadow:0 1px 4px rgba(0,0,0,0.3)`;

            addIcon.addEventListener('click', (e) => {
                e.preventDefault();
                e.stopPropagation();
                const url = new URL(img.src, location.href).href;
                if (groups.my.some(item => item.url === url)) {
                    showToast('图片已存在');
                } else {
                    groups.my.push({ url, desc: img.alt || '图片描述' });
                    saveGroup('my');
                    showToast('已添加到“我的”组');
                }
                addIcon.remove();
                delete img.dataset.emojiCollector;
            });

            parent.appendChild(addIcon);
            parent.addEventListener('mouseleave', () => {
                addIcon.remove();
                delete img.dataset.emojiCollector;
            }, { once: true });
        });
    }

    // ==================== 初始化 ====================
    function init() {
        const emojiBtn = document.querySelector('button.nb-editor-btn[data-nb-editor-emoji="1"]');
        if (!emojiBtn) return setTimeout(init, 500);
        if (!newBtn && emojiBtn.parentNode) {
            emojiBtn.parentNode.insertBefore(createEditorButton(emojiBtn), emojiBtn);
        }
        if (!window.__emojiCollectorInit) {
            window.__emojiCollectorInit = true;
            initPostImageCollector();
        }
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', init);
    } else {
        init();
    }
})();
